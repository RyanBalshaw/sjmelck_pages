{{- $context := . -}} {{- $imagePath := path.Join "images" .Page.File.TranslationBaseName .Destination -}} {{- with resources.GetMatch $imagePath -}} {{- $image := . -}} {{- $isGIF := eq .MediaType.SubType "gif" -}} {{- $src := .RelPermalink -}} {{- $srcset := slice -}} {{- if not $isGIF -}} {{- range slice 480 800 1200 -}} {{- if ge $image.Width . -}} {{- $resized := $image.Resize (printf "%dx q82" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $resized.RelPermalink .) -}} {{- end -}} {{- end -}} {{- if and (lt $image.Width 1200) (ne $image.Width 480) (ne $image.Width 800) -}} {{- $srcset = $srcset | append (printf "%s %dw" $image.RelPermalink $image.Width) -}} {{- end -}} {{- if gt (len $srcset) 0 -}} {{- $preferred := $image -}} {{- if gt $image.Width 800 }}{{ $preferred = $image.Resize "800x q82" }}{{ end -}} {{- $src = $preferred.RelPermalink -}} {{- end -}} {{- end -}} {{- $caption := $context.Title | default $context.Text -}} {{- $generic := or (eq (lower $caption) "alt text") (eq (lower $caption) "image alt text") -}} {{- else -}} {{- end -}}
